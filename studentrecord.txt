public class studentrecord {

    public Student__c stu {get; set;}
    public List<Student__c> studentList {get; set;}

    // Variable to check edit mode
    public Boolean isEdit {get; set;}

    public studentrecord() {

        stu = new Student__c();

        isEdit = false;

        loadStudents();
    }

    // Load all students
    public void loadStudents() {

        studentList = [SELECT Id, Name, Roll__c,
                       Class__c, Mobile_No__c
                       FROM Student__c
                       ORDER BY CreatedDate DESC];
    }

    // Save or Update student
    public void saveStudent() {

        if(isEdit == false) {

            insert stu;

        } else {

            update stu;

            isEdit = false;
        }

        // Clear form
        stu = new Student__c();

        loadStudents();
    }

    // Edit student
    public void editStudent() {

        String studentId = ApexPages.currentPage()
                          .getParameters()
                          .get('editId');

        stu = [SELECT Id, Name, Roll__c,
               Class__c, Mobile_No__c
               FROM Student__c
               WHERE Id = :studentId];

        isEdit = true;
    }

    // Delete student
    public void deleteStudent() {

        String studentId = ApexPages.currentPage()
                          .getParameters()
                          .get('delId');

        Student__c s = [SELECT Id
                        FROM Student__c
                        WHERE Id = :studentId];

        delete s;

        loadStudents();
    }

    // Cancel edit
    public void cancelEdit() {

        stu = new Student__c();

        isEdit = false;
    }
}
